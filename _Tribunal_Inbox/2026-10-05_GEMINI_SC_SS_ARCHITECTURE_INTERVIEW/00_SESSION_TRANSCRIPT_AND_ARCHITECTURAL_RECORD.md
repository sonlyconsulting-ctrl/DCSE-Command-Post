            {"heading": "Cormorant Garamond", "body": "Merriweather"}
        ],
        "allowed_layouts": ["Cinematic_Scroll", "Bento_Box_Asymmetric", "Hero_Full_Bleed"]
    }
}

def generate_type_scale(base, ratio, levels=5):
    """Calculates H5 up to H1 mathematically."""
    scale = {}
    current_size = base
    for i in range(levels):
        scale[f"H{5-i}"] = round(current_size, 2)
        current_size *= ratio
    scale["body"] = base
    return scale

def assign_layout_and_type(entity_id, page_id, ledger_path="architecture_ledger.json"):
    """
    Checks the inventory ledger to find an unused font pairing and layout archetype for a specific page.
    """
    with open(ledger_path, 'r') as file:
        state_ledger = json.load(file)
        
    tokens = BRAND_TOKENS[entity_id]
    type_scale = generate_type_scale(tokens["base_font_px"], tokens["scale_ratio"])
    
    for font in tokens["font_pairings"]:
        for layout in tokens["allowed_layouts"]:
            config_id = f"{font['heading']}-{layout}"
            
            if config_id not in state_ledger.get(entity_id, []):
                if entity_id not in state_ledger:
                    state_ledger[entity_id] = []
                state_ledger[entity_id].append(config_id)
                with open(ledger_path, 'w') as file:
                    json.dump(state_ledger, file)
                return {
                    "fonts": font,
                    "scale": type_scale,
                    "layout_archetype": layout,
                    "status": "Assigned & Locked"
                }
